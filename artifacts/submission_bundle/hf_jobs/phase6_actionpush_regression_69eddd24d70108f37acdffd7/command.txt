-lc
set -euxo pipefail
mkdir -p /workspace
cd /workspace
python - <<'PY'
import os
import urllib.request
import zipfile
url = os.environ['HF_REPO_ZIP_URL']
zip_path = '/workspace/repo.zip'
urllib.request.urlretrieve(url, zip_path)
with zipfile.ZipFile(zip_path) as zf:
    zf.extractall('/workspace')
PY
cd /workspace/physio-support-openenv-main
python - <<'PY'
from pathlib import Path
src = Path('requirements.txt').read_text(encoding='utf-8').splitlines()
out = []
for line in src:
    stripped = line.strip()
    if not stripped or stripped.startswith('#') or stripped.startswith('torch'):
        continue
    out.append(line)
Path('requirements_notorch.txt').write_text('\n'.join(out) + '\n', encoding='utf-8')
PY
python -c "import torch; print('torch', torch.__version__, 'cuda', torch.cuda.is_available())"
python -m pip install --upgrade pip
python -m pip install -r requirements_notorch.txt

python - <<'PY'
from pathlib import Path
